# chr 목록(모두 chunk로 나눔; chrM/chrY도 여기 포함해도 됨)
printf "chr%s\n" {1..22} X Y M \
| xargs -I{} -P 26 bash -c '
set -euo pipefail
chr="{}"

# 입력 파일 자동 탐색 (파일명 규칙 흔들려도 대응)
in=$(ls "'$IN05'"/CPC.HPRC.Phase1.*${chr}*.split.renamed.vcf.gz 2>/dev/null | head -n1)
if [[ -z "${in:-}" ]]; then
  echo "[skip] missing input for $chr" >&2
  exit 0
fi

outdir="'$CHUNKROOT'/$chr"
mkdir -p "$outdir"

hdr="$outdir/header.vcf"
echo "[chunk] $chr : write header" >&2
bcftools view -h "$in" > "$hdr"

echo "[chunk] $chr : split body into ${CHUNKSIZE}-line chunks" >&2
bcftools view -H "$in" \
| split -l '"$CHUNKSIZE"' -d -a 5 - "$outdir/body_"

# 메타 저장(나중에 merge 때 유용)
echo "$in" > "$outdir/input.path"
'

set -euo pipefail

ROOT=/BDATA/smkim/pangenome/graph/CPC/vcfbub
CHUNKROOT=$ROOT/06.chunks_bychr
WAVEROOT=$ROOT/07.wave_chunks
LOG=$ROOT/log/07.wave_chunks

mkdir -p "$WAVEROOT" "$LOG"

export WAVE_I=1000
export WAVE_T=12        # chunk 하나당 thread
export BGZIP_T=2
export CHUNK_JOBS=16   # 동시에 돌릴 chunk 개수(=병렬 수)

# 모든 body chunk를 작업 목록으로 만들어 병렬 실행
find "$CHUNKROOT" -type f -name "body_*" -print \
| shuf \
| xargs -I{} -P "$CHUNK_JOBS" bash -c '
set -euo pipefail
body="{}"
chr=$(basename "$(dirname "$body")")
hdr="'"$CHUNKROOT"'/$chr/header.vcf"

outdir="'"$WAVEROOT"'/$chr"
mkdir -p "$outdir"

base=$(basename "$body")         # body_00000
tmpvcf="$outdir/${base}.vcf"     # header+body 합친 임시 VCF
outgz="$outdir/${base}.wave.vcf.gz"
stderr="'"$LOG"'/${chr}.${base}.stderr.log"

# 이미 성공 산출물이 있으면 skip(재시도 편하게)
if [[ -s "$outgz" ]]; then
  exit 0
fi

cat "$hdr" "$body" > "$tmpvcf"

vcfwave -I '"$WAVE_I"' -t '"$WAVE_T"' "$tmpvcf" \
  1> >(bgzip -@ '"$BGZIP_T"' -c > "$outgz") \
  2>> "$stderr"

rm -f "$tmpvcf"
'

comm -23 "$WORK/KPP.small.key.tsv" "$WORK/CPCHPRC.small.key.tsv" \
  > "$RES/KPP.unique.small.key.tsv"

# CPC/HPRC unique
comm -13 "$WORK/KPP.small.key.tsv" "$WORK/CPCHPRC.small.key.tsv" \
  > "$RES/CPCHPRC.unique.small.key.tsv"

# count
{
  echo -n "KPP total small: "
  wc -l < "$WORK/KPP.small.key.tsv"

  echo -n "CPCHPRC total small: "
  wc -l < "$WORK/CPCHPRC.small.key.tsv"

  echo -n "shared small: "
  wc -l < "$RES/shared.small.key.tsv"

  echo -n "KPP unique small: "
  wc -l < "$RES/KPP.unique.small.key.tsv"

  echo -n "CPCHPRC unique small: "
