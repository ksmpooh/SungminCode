#!/usr/bin/env bash
set -euo pipefail

BASE=/BDATA/smkim/pangenome/graph/final_vs_newkpp
RAW=$BASE/00.raw
WORK=$BASE/01.work/small
RES=$BASE/02.result/small

mkdir -p "$RAW" "$WORK" "$RES"

KPP=/BDATA/smkim/pangenome/graph/KPPD/new_wave/09.concat/kor132.chm13v2.small.allchr.vcf.gz
CPC=/BDATA/smkim/pangenome/graph/CPC/vcfbub/12.concat/CPC.HPRC.Phase1.small.allchr.vcf.gz

cp -f "$KPP" "$RAW/"
cp -f "$KPP".tbi "$RAW/" 2>/dev/null || true
cp -f "$CPC" "$RAW/"
cp -f "$CPC".tbi "$RAW/" 2>/dev/null || true

KPP_LOCAL=$RAW/$(basename "$KPP")
CPC_LOCAL=$RAW/$(basename "$CPC")

bcftools query -f '%CHROM\t%POS\t%REF\t%ALT\n' "$KPP_LOCAL" \
  | sort -u > "$WORK/KPP.small.key.tsv"

bcftools query -f '%CHROM\t%POS\t%REF\t%ALT\n' "$CPC_LOCAL" \
  | sort -u > "$WORK/CPCHPRC.small.key.tsv"

comm -12 "$WORK/KPP.small.key.tsv" "$WORK/CPCHPRC.small.key.tsv" \
  > "$RES/shared.small.key.tsv"

comm -23 "$WORK/KPP.small.key.tsv" "$WORK/CPCHPRC.small.key.tsv" \
  > "$RES/KPP.unique.small.key.tsv"

comm -13 "$WORK/KPP.small.key.tsv" "$WORK/CPCHPRC.small.key.tsv" \
  > "$RES/CPCHPRC.unique.small.key.tsv"

{
  echo -n "KPP total small: "
  wc -l < "$WORK/KPP.small.key.tsv"

  echo -n "CPCHPRC total small: "
  wc -l < "$WORK/CPCHPRC.small.key.tsv"

  echo -n "shared small: "
  wc -l < "$RES/shared.small.key.tsv"

  echo -n "KPP unique small: "
  wc -l < "$RES/KPP.unique.small.key.tsv"

  echo -n "CPCHPRC unique small: "
  wc -l < "$RES/CPCHPRC.unique.small.key.tsv"
} | tee "$RES/summary.small.txt"
