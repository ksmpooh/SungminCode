#필요없는 리드 seqkit로 없애는 법
conda activate seqkit_env
FASTQ=PG240004_LCL_ULK.merge.fastq.gz
seqkit seq --threads 100 --min-qual 15 $FASTQ > ${FASTQ%%.fastq.gz}.15q.fastq
pigz ${FASTQ%%.fastq.gz}.10q.fastq

seqkit seq --min-len 10000 $FASTQ > ${FASTQ%%.fastq.gz}.min10kb.fastq

#롱리드 품질 Nanoplot으로 확인하는 법
conda activate nanoplot_env
for FILE in $(ls | grep fastq.gz )
do echo $FILE
FILE2=${FILE%%.fastq.gz}
echo $FILE2
NanoPlot --verbose --raw --huge --tsv_stats --threads 100 --outdir ${FILE2}_nanoplot --prefix ${FILE2}_ --fastq $FILE
done
