#!/usr/bin/env bash
set -euo pipefail

BASE=/ADATA/smkim/pangenome/bandage/haplotype/C4AB/scaffolding/KPPD
IN=$BASE/01.input
OUT=$BASE/02.screen
LIST=$IN/fasta.list
QUERY=$IN/c4ab_screen_query.fa

mkdir -p "$OUT/paf" "$OUT/hit_ids" "$OUT/log"

run_one() {
  fa="$1"
  bn=$(basename "$fa" .fa)
  paf="$OUT/paf/${bn}.paf"
  ids="$OUT/hit_ids/${bn}.ids.txt"
  log="$OUT/log/${bn}.log"

  echo "[screen] $bn"

  minimap2 -x asm20 -N 50 -p 0.1 --secondary=yes \
    "$fa" "$QUERY" > "$paf" 2> "$log"

  awk 'BEGIN{FS="\t"} $10>=500 {print $6}' "$paf" | sort -u > "$ids"
}
export -f run_one
export OUT QUERY

cat "$LIST" | xargs -I{} -P 16 bash -c 'run_one "$@"' _ {}
echo "[done]"
EOF

#!/usr/bin/env bash
set -euo pipefail

BASE=/ADATA/smkim/pangenome/bandage/haplotype/C4AB/scaffolding/KPPD
IN=$BASE/01.input
SUB=$BASE/03.subset/fasta_chr6
OUT=$BASE/04.map
GRAPH=$IN/c4ab.gfa

mkdir -p "$OUT/gaf_chr6" "$OUT/log_chr6"

run_one() {
  fa="$1"
  bn=$(basename "$fa" .c4ab_chr6.fa)
  gaf="$OUT/gaf_chr6/${bn}.gaf"
  log="$OUT/log_chr6/${bn}.log"

  echo "[graphaligner-chr6] $bn"

  if [ ! -s "$fa" ]; then
    : > "$gaf"
    echo "empty_subset" > "$log"
    exit 0
  fi

  if ! grep -q '^>' "$fa"; then
    : > "$gaf"
    echo "no_fasta_header" > "$log"
    exit 0
  fi

  GraphAligner \
    -g "$GRAPH" \
    -f "$fa" \
    -a "$gaf" \
    -x vg \
    -t 6 \
    > "$log" 2>&1
}
export -f run_one
export GRAPH OUT

ls "$SUB"/*.c4ab_chr6.fa | xargs -I{} -P 50 bash -c 'run_one "$@"' _ {}
echo "[done]"
EOF
